import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import {
  getInterventionOptions, listInterventions, getIntervention, createIntervention,
  updateIntervention, deleteIntervention, restoreIntervention, addFollowup, listDueFollowUps,
} from '../services/interventionService';

export const useInterventionOptions = () =>
  useQuery({
    queryKey: ['intervention-options'],
    queryFn: getInterventionOptions,
    staleTime: Infinity, // fixed lists
  });

export const useInterventions = (studentId, classId, trashed) =>
  useQuery({
    queryKey: ['interventions', studentId, classId || '', !!trashed],
    queryFn: () => listInterventions({ studentId, classId, trashed }),
    enabled: !!studentId,
  });

export const useIntervention = (id) =>
  useQuery({
    queryKey: ['intervention', String(id)],
    queryFn: () => getIntervention(id),
    enabled: !!id,
  });

export const useDueFollowUps = () =>
  useQuery({
    queryKey: ['follow-ups-due'],
    queryFn: listDueFollowUps,
  });

// Any change to an intervention affects the student lists, the detail and the due list.
const REFRESH = ['interventions', 'intervention', 'follow-ups-due'];

function useRefreshingMutation(mutationFn) {
  const qc = useQueryClient();
  return useMutation({
    mutationFn,
    onSuccess: () => Promise.all(REFRESH.map((k) => qc.invalidateQueries({ queryKey: [k] }))),
  });
}

export const useCreateIntervention = () => useRefreshingMutation(createIntervention);
export const useUpdateIntervention = () => useRefreshingMutation(updateIntervention);
export const useDeleteIntervention = () => useRefreshingMutation(deleteIntervention);
export const useRestoreIntervention = () => useRefreshingMutation(restoreIntervention);

export function useAddFollowup() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: addFollowup,
    onSuccess: (data, vars) => {
      qc.setQueryData(['intervention', String(vars.id)], data); // the response is the fresh detail
      qc.invalidateQueries({ queryKey: ['interventions'] });
      qc.invalidateQueries({ queryKey: ['follow-ups-due'] });
    },
  });
}

// Mounted only after the options exist, so the selects start with the right value.
function InterventionForm({ options, studentId, classes, intervention, initial, onClose, onSaved }) {
  const isEdit = !!intervention;
  const create = useCreateIntervention();
  const update = useUpdateIntervention();
  const mutation = isEdit ? update : create;

  const { register, handleSubmit, setError, formState: { errors } } = useForm({
    resolver: zodResolver(schema),
    defaultValues: isEdit
      ? {
          concern_type: intervention.concern_type,
          concern: intervention.concern ?? '',
          intervention_type: intervention.intervention_type,
          intervention_date: intervention.intervention_date,
          notes: intervention.notes ?? '',
        }
      : {
          class_assignment_id: initial?.class_assignment_id ? String(initial.class_assignment_id) : '',
          concern_type: initial?.concern_type ?? '',
          concern: initial?.concern ?? '',
          intervention_type: '',
          intervention_date: todayLocal(),
          follow_up_date: '',
          notes: '',
        },
  });

  const onSubmit = (v) => {
    const common = {
      concern_type: v.concern_type,
      concern: v.concern.trim(),
      intervention_type: v.intervention_type,
      intervention_date: v.intervention_date,
      notes: v.notes?.trim() ? v.notes.trim() : null,
    };

    mutation.mutate(
      isEdit
        ? { id: intervention.id, ...common }
        : {
            studentId,
            class_assignment_id: v.class_assignment_id ? Number(v.class_assignment_id) : null,
            follow_up_date: v.follow_up_date || null,
            ...common,
          },
      {
        onSuccess: (result) => {
          onSaved?.(result);
          onClose();
        },
        onError: (err) => applyServerErrors(err, setError, FIELDS),
      }
    );
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="space-y-4">
      <p className="rounded-md bg-gray-50 p-3 text-xs text-gray-600">

        here is sent to anyone.
      </p>

      {!isEdit && (
        <FormField label="Class" error={errors.class_assignment_id?.message} hint="Leave as General if this is not about one class.">
          <select className={inputCls} {...register('class_assignment_id')}>
            <option value="">General (not tied to a class)</option>
            {classes.map((c) => (
              <option key={c.id} value={c.id}>{c.label} - {c.subject}</option>
            ))}
          </select>
        </FormField>
      )}

      <div className="grid grid-cols-2 gap-3">
        <FormField label="Concern type" error={errors.concern_type?.message}>
          <select className={inputCls} {...register('concern_type')}>
            <option value="">Choose...</option>
            {options.concern_types.map((o) => <option key={o.value} value={o.value}>{o.label}</option>)}
          </select>
        </FormField>
        <FormField label="Intervention" error={errors.intervention_type?.message}>

        </FormField>
      </div>

      <FormField label="Concern" error={errors.concern?.message}>
